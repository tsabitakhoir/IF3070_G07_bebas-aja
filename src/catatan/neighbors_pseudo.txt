Berarti perlu fungsi untuk copy state terus update state

fungsi copy_state(state):
    return [dict(v) dari each v in state] (v as in vehicle)

fungsi update(state, updates): (updates -> list of dict isinya state yang berubah)
    state baru = copy_state(state)
    for i, v in updates.items():
        state baru[i] = v
    return state baru
